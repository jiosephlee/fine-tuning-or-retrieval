all_rules   = []
for t in tqdm.tqdm(range(NUM_ENS)):
    # optionally subsample features if context window exceeded:
    sub_feat   = random_subset(features,
                               size=BAG_SIZE or int(np.sqrt(len(features))))
    prompt     = render_prompt(prompt_tpl,
                               feat_subset=sub_feat,
                               seed=t)
    raw_rules   = get_rules(prompt)
    
    # translate human phrasing → python lambda functions:
    py_rules   = {cls:[compile_rule(r) for r in lst]
                  for cls,lst in raw_rules.items()}
    
    all_rules.append(py_rules)
