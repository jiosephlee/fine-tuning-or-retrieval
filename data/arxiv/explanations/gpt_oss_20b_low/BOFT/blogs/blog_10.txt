def cayley_from_skew(Q):
    """Return R = (I + Q)(I - Q)^{-1} where Q is skew-symmetric."""
    I = torch.eye(Q.size(0), device=Q.device, dtype=Q.dtype)
    # Avoid division by zero when Q is identity – not happening here.
    return torch.linalg.solve(I - Q, I + Q)

class ButterflyBlock(nn.Module):
    """
    A single butterfly component with batch-size‐independent parameters.
    'size' must be even; we optimise 'b'×'b' sub‑blocks inside.
    """
    def __init__(self, size: int, block: int = 2):
        super().__init__()
        assert size % 2 == 0 and block <= size
        self.size = size
        self.block = block
        
        # Declare skew matrices for each b×b pair inside diag(b,b,…)
        num_blocks = self.size // self.block
        # Q will eventually be block-diagonal composed of num_blocks submatrices
        # Each submatrix holds its own skew entries.
        self.skews = nn.Parameter(torch.zeros(num_blocks, 
                                              self.block,
                                              self.block))
        
    def forward(self):
        # Build full skew matrix from batch of small skews
        # Shape -> num_blocks × block × block → batch × size × size
        skew_full = torch.zeros(self.size, self.size,
                                device=self.skews.device,
                                dtype=self.skews.dtype)
        
        for i in range(self.size // self.block):
            idx = slice(i*self.block, (i+1)*self.block)
            sub = self.skews[i]
            # Make it skew-symmetric: S - Sᵀ
            skew_sub = sub - sub.t()
            skew_full[idx,:][:,idx] = skew_sub
        
        return cayley_from_skew(skew_full)
```

### Why this design?
- Each sub‐block `(block × block)` keeps calculations cheap.
- `skews` are stored per block; this automatically gives us the desired sparsity pattern.
- The forward call returns an *augmented* orthogonal matrix constructed from all subblocks.

## 4. Assembling the Whole BOFT Layer

Now we chain multiple `ButterflyBlock`s together according to how many stages you want (`m`). Let us also expose an optional `block_size` hyperparameter (`b`).

```python
class BOFTLayer(nn.Module):
    """
    Implements R w^0 x where R is product of m butterfly blocks.
    
    Args:
        orig_weight: Pretrained frozen weight matrix W^0     shape [out_dim, in_dim]
                     – *Note*: we expect it transposed in usage!
        m: number of butterfly stages (log₂(out_dim/block_size))
           smaller m => larger blocks => fewer params.
        block_size: b >= 2 – controls granularity inside each block.
    """
    
    def __init__(self, orig_weight: torch.Tensor,
                 m: int,
                 block_size: int = 2,
                 dropout_rate_block: float = 0.,
                 dropout_rate_sub: float   = 0.):

        super().__init__()
        
        out_dim, in_dim   = orig_weight.shape
        assert out_dim % block_size == 0 and \
               out_dim % (block_size * (2**m)) == 0
        
        self.orig_w   = nn.Parameter(orig_weight.clone(), requires_grad=False)
        
        # Create m butterfly layers (each has same size)
        self.blocks   = nn.ModuleList(
            [ButterflyBlock(size=out_dim,
                            block=block_size)
             for _ in range(m)]
        )
        
        # Dropout hyperparams
        self.drop_block      = dropout_rate_block
        self.drop_subblock   = dropout_rate_sub
    
    def forward(self, x):
        """
        x shape [batch_size, out_dim] -- expecting pre-transformed activations
        
           if using GELU( W^T x ), call with x such that 
           output shape is exactly [batch, out_dim].
           
           Useful when you want to replace only one linear layer inside MLP/FFN head.
           
           Returning y : [batch, in_dim] (since W^T acts).
           But typical usage is y^T W^T → dropping dims symmetrically.
           
           Decision:
               >>> z = BFT_layer(x.T).T 
               Swaps dims appropriately.
           
           For clarity we keep y.shape same as input iff orig_w.transpose() used elsewhere.
       
       """
       
       # multiplicative dropout over whole blocks:
       if self.training and self.drop_block > 0:
           mask_block   = torch.bernoulli(
               torch.full((len(self.blocks),),
                          1 - self.drop_block,
                          device=x.device))
           active_blocks = [blk if m else IdentityBlock(blk.size)
                            for blk,m in zip(self.blocks, mask_block)]
       else:
           active_blocks      = list(self.blocks)

       # Apply blocks sequentially on the left side:
       R_mat              = None
       for blk in active_blocks[::-1]:   # rightmost first for stacking on left?
           Xblk              = blk()     # orthogonal matrix from trainable params
 
           # multiplicative dropout on subblocks:
           if self.training and self.drop_subblock > 0:
               mask_sb          = torch.bernoulli(
                   torch.full((Xblk.shape[0]//blk.block,),
                              1 - self.drop_subblock,
                              device=Xblk.device))
               Xblk              *= mask_sb[:, None]     # broadcast across rows
   
           R_mat              = Xblk if R_mat is None else Xblk @ R_mat
        
       # NOTE: Ruby's @ operator multiplies from left-to-right;
       # Here we want R * Wᵀ; We compute Wᵀ first then multiply by R.
       y                  = R_mat @ torch.transpose(self.orig_w, 0,1)
       
       return y @ x.T     # output shape [batch,in_dim]
```

> **Note** – Boilerplate above used an “IdentityBlock” util that simply returns an identity when a component is dropped out; you can implement it as:

```python
class IdentityBlock(nn.Module):
    def __init__(self,size): super().__init__()
    def forward(self): return torch.eye(size, device='cuda')
```

But easier you may simply bypass passing that component – see below.

### Weight initialization (`identity_init`)

Since all skews are zero at construction time (`nn.Parameter(zeros)`), every small subblock starts as an identity matrix (`Q=0 → R=I`). Thus the overall product starts as identity too—exactly what we want (`R=Wᵀ`). No special init procedure is required beyond the zero initialization already done by PyTorch on parameters.
