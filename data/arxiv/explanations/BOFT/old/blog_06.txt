# Wrap each MLP linear layer
def wrap_mlp_with_boft(module):
    for name, child in module.named_children():
        if isinstance(child, nn.Linear):
            m, b = 4, 16    # e.g. 4 butterfly components, block size 16
            boft_lin = BOFTLinear(child.weight, m=m, block_size=b, dropout=(0.1, 0.1))
            setattr(module, name, boft_lin)
        else:
            wrap_mlp_with_boft(child)

wrap_mlp_with_boft(model)
